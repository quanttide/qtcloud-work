# 工作流＝一串有序的步骤；每步写着谁做、怎么算完
name: devops-release
description: 给 qtcloud-work CLI 走一次预发布 ……
steps:
- name: version          # 定版本
  executor: agent          # 这一步谁做
  criteria:
  - executor: rule
    description: 版本文件在
    path: apps/qtcloud-work/src/cli/Cargo.toml
  - executor: rule
    description: 版本号写法在档位上
    run: grep -qE '^version = ...' Cargo.toml
  - executor: agent
    description: 版本号、CHANGELOG、Cargo.toml 三处一致

- name: gate            # 质量门禁，与 CI 同参数
  executor: agent
  criteria:
  - { executor: rule, run: cargo fmt --check }
  - { executor: rule, run: cargo clippy --locked --all-targets -- -D warnings }
  - { executor: rule, run: cargo test --locked }

- name: confirm         # 人点头：这一步的判据主要是人
  executor: human
  criteria:
  - executor: human
    description: 创始人确认版本号与变更范围，放行这次预发布

# 跑起来：qtcloud-work task --new devops-release --workflow devops-release，再 --next 一条一条走
