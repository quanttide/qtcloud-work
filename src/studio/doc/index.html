# 静态骨架（YAML）
name: 竞品分析
version: 1.3
inputs:
  topic: string
steps:
  - id: input            type: input
  - id: split            type: agent      model: light
  - id: gate             type: condition  expr: complexity > 3
  - id: retrieve_docs    type: agent      model: light
  - id: retrieve_reviews type: agent      model: light
  - id: approve          type: human_approval
  - id: output           type: output     path: report.md

# 动态环节（JS DSL）
parallel(["retrieve_docs", "retrieve_reviews"])
  .map(agent({ schema: Finding[] }))
  .then(agent("build_matrix"))
  .then(approve({ owner: "human" }))
  .then(output("report.md"))
